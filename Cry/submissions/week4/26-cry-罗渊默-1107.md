```python
# ========== 0. 导入依赖 ==========
import sys
from collections import Counter          ## 统计字母出现次数，算 IC 用
from hashlib import md5                  ## 校验 key
from math import log                     ## 似然取对数，防止连乘下溢
from pathlib import Path                 ## 定位脚本目录，稳健读文件

sys.stdout.reconfigure(encoding="utf-8")  ## Windows 控制台默认 GBK，不改编码打印中文会崩

## ========== 1. 读入密文 ==========
HERE = Path(__file__).parent                                    ## 取脚本所在目录，不依赖当前工作目录
ct = (HERE / "cipher").read_text(encoding="utf-8").strip()      ## 一次性读入全部密文，去掉尾部换行
print("[i] 密文长度:", len(ct))                                  ## 自检：长度对不上说明文件读错了

## ========== 2. 用重合指数 IC 找周期 ==========
def ic(s):
    """重合指数：任取两个字符，它们相同的概率。
       英文文本 ≈ 0.066，随机字母 ≈ 0.038 —— 这个差距就是找周期的依据"""
    n = len(s)
    if n < 2:
        return 0.0                                              ## 少于 2 个字符凑不成对，返回 0 防除零
    c = Counter(s)                                              ## 统计本组里每个字母出现多少次
    return sum(v * (v - 1) for v in c.values()) / (n * (n - 1))
    ## 分子：每个字母能组成的有序对 v*(v-1)
    ## 分母：总的有序对数 n*(n-1)
    ## 比值就是"任取两个字符相同"的概率


best_t, best_v = 0, 0.0
for t in range(1, 101):                                         ## 题干 a*b<100 → 周期最多 99，多扫到 100 防边界
    avg = sum(ic(ct[r::t]) for r in range(t)) / t
    ## ct[r::t]：从第 r 个字符起、每隔 t 个取一个
    ## 相当于把密文按"位置模 t"切成 t 组
    ## 每组算一次 IC，再取平均：周期猜对了，组内是纯英文，平均 IC 会明显抬高
    if avg > best_v:
        best_t, best_v = t, avg                                  ## 记录目前最好的周期和对应的 IC
L = best_t                                                       ## 最终周期
print(f"[+] 周期 = {L}（平均组内 IC = {best_v:.4f}）")            ## 实跑：77，IC=0.0649，非常接近英文 0.066

## ========== 3. 英文字母频率表 ==========
FREQ = [8.167, 1.492, 2.782, 4.253, 12.702, 2.228, 2.015, 6.094, 6.966, 0.153, 0.772,
        4.025, 2.406, 6.749, 7.507, 1.929, 0.095, 5.987, 6.327, 9.056, 2.758, 0.978,
        2.360, 0.150, 1.974, 0.074]
## 上表是 a~z 的出现频率（百分比），来源是标准英文语料统计
f = [x / 100 for x in FREQ]                                      ## 转成 0~1 的概率，后面当似然用

## ========== 4. 按周期分组计数 ==========
counts = [[0] * 26 for _ in range(L)]                            ## counts[r][c]：第 r 组里字母 c 出现了几次
for i, ch in enumerate(ct):
    counts[i % L][ord(ch) - 97] += 1
    ## ord(ch)-97 把 a~z 映射成 0~25
    ## i % L 决定这个字符属于哪一组
    ## 一遍 O(n) 遍历就把 10.8 万个字符分好组了

## ========== 5. 逐组拟合仿射映射 c = A*p + B ==========
models = []                                                      ## 每组的拟合结果 (分数, A, B)
for r in range(L):                                               ## 第一层：遍历 77 组
    cnt = counts[r]                                              ## 取出第 r 组的字母计数
    best = None
    for A in range(26):                                          ## 第二层：枚举乘数 A（0~25）
        for B in range(26):                                      ## 枚举加数 B（0~25），共 676 种组合
            s = 0.0                                              ## 这个模型的似然得分
            ok = True
            for c in range(26):                                  ## 第三层：遍历 26 个密文字母
                k = cnt[c]
                if not k:
                    continue                                     ## 本组没出现过这个密文字母，跳过
                tot = sum(f[p] for p in range(26) if (A * p + B) % 26 == c)
                ## tot = "能被加密成密文字母 c 的所有明文字母"的概率之和
                ## 注意写法允许 A 不可逆（多个 p 映射到同一个 c 时会有几项相加）
                if tot <= 0:
                    ok = False                                   ## 没有任何明文能映射到 c，这个模型不可能
                    break
                s += k * log(tot)                                ## 对数似然：概率连乘会下溢，取对数变加法
            if ok and (best is None or s > best[0]):
                best = (s, A, B)                                 ## 保留得分最高的那组参数
    models.append(best)

## ========== 6. 反推题干里的 a、b ==========
As = [m[1] for m in models]                                      ## 77 组各自的 A
Bs = [m[2] for m in models]                                      ## 77 组各自的 B
perA = next(p for p in range(1, L) if all(As[r] == As[(r + p) % L] for r in range(L)))
perB = next(p for p in range(1, L) if all(Bs[r] == Bs[(r + p) % L] for r in range(L)))
## next(...)：找最小的 p，使得序列每隔 p 位就完全重复 —— 这就是序列的周期
## A 的周期 = 题干的 a，B 的周期 = 题干的 b
print(f"[+] A 序列周期 = {perA} | B 序列周期 = {perB}   （题干 a={perA}, b={perB}）")
## 实跑：11 和 7，乘积 77 < 100，与周期检测互相印证

## ========== 7. 解密 ==========
out = []
amb = 0                                                          ## 统计"有歧义"的位置：A 与 26 不互素时会有多解
for i, ch in enumerate(ct):
    c = ord(ch) - 97                                             ## 当前密文字母，转成 0~25
    _, A, B = models[i % L]                                      ## 取出该位置所属组的参数（下划线忽略分数）
    cands = [p for p in range(26) if (A * p + B) % 26 == c]      ## 反解：所有满足 A*p+B≡c 的明文字母
    if len(cands) > 1:
        amb += 1                                                 ## 候选不唯一，说明这一位有歧义
    out.append(chr(97 + max(cands, key=lambda x: f[x])))
    ## 有多个候选时，挑英文字母频率最高的那个（最可能是真明文）
plain = "".join(out)
print(f"[+] 歧义位置: {amb} 处 | 明文开头: {plain[:60]}")
## 实跑：0 处歧义（77 组的 A 全是与 26 互素的奇数），开头是《德米安》的英文首句

## ========== 8. 用 md5 在明文里搜出 key ==========
target = "196cf7098c7ea6e3e4d03691fb9d4f58"                     ## 题干给的 md5
suffix = b"How_Interesting_the_Cryptography_Is"                  ## 题干给的后缀，注意是 bytes
key = None
pos = -1
for ln in range(1, 21):                                          ## 长度从 1 递增到 20（保证找到最短的匹配）
    for i in range(len(plain) - ln + 1):                         ## 滑窗上界，防止越界
        sub = plain[i:i + ln]                                    ## 截取一个候选子串
        if md5(sub.encode() + suffix).hexdigest() == target:     ## 拼上后缀算 md5，与目标比对
            key = sub
            pos = i
            break
    if key:
        break                                                    ## 命中就跳出外层长度循环

## ========== 9. 输出结果 ==========
if key:
    print(f"[+] key = {key}（原文明文中位置 {pos}）")
    print(f"[+] 上下文: ...{plain[pos - 40:pos]}[{key}]{plain[pos + len(key):pos + len(key) + 40]}...")
    ## 打印上下文，人工确认这个 key 确实是一个自然短语，而不是 md5 偶然碰撞
    print(f"[+] flag = moectf{{{key}}}")                         ## f-string 里 {{ }} 用来输出花括号
else:
    print("[-] 没找到 key")                                      ## 兜底：长度范围内没撞上
```
