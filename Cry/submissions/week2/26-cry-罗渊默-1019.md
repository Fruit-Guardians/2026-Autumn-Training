```python
import gmpy2
from Crypto.Util.number import long_to_bytes

def commmon_modulus_attack(n1,e1,e2,c1,c2):
    g,x,y = gmpy2.gcdext(e1,e2)
    if g != 1:
        print("非互素，滚木")
        return none
    if x< 0:
        x = -x
        c1 = gmpy2.invert(c1,n1)
    if y< 0:
        y= -y
        c2 = gmpy2.invert(c2,n1)

    m = (pow(c1,x,n1)*pow(c2,y,n1))%n1
    return long_to_bytes(m)

result = commmon_modulus_attack(n1,e1,e2,c1,c2)
if result is not None:
    print(result.decode('utf-8'))
else:
    print("滚木")
```
